-- 00) 테이블 생성

-- tbl_member
DROP TABLE IF EXISTS tbl_member;
CREATE TABLE tbl_member (
    Member_id INT PRIMARY KEY,
    Member_name VARCHAR(45),
    Member_identity VARCHAR(45),
    Member_grade VARCHAR(45),
    Member_addr VARCHAR(45),
    Member_phone VARCHAR(45)
);

-- tbl_book
DROP TABLE IF EXISTS tbl_book;
CREATE TABLE tbl_book (
    Book_code INT PRIMARY KEY,
    Classification_ID INT,
    book_author VARCHAR(45),
    book_name VARCHAR(45),
    publisher VARCHAR(45),
    isrental CHAR(1)
);

-- tbl_rental
DROP TABLE IF EXISTS tbl_rental;
CREATE TABLE tbl_rental (
    Rental_id INT,
    Book_code INT,
    Member_id INT
);

-- 1) tbl_member의 Member_id 에 Primary key 제약 조건 설정 (이미 생성 시 지정했지만 따로 분리할 경우)
-- 생략 가능. 이미 PRIMARY KEY 지정됨

-- 2) tbl_book의 Book_code 에 Primary key 제약 조건 설정 (생성 시 지정됨)

-- 3) tbl_rental의 Book_code 에 FK 설정 (ON DELETE CASCADE)
ALTER TABLE tbl_rental
ADD CONSTRAINT fk_Book_code
FOREIGN KEY (Book_code)
REFERENCES tbl_book(Book_code)
ON DELETE CASCADE;

-- 4) tbl_rental의 Member_id 에 FK 설정 (ON DELETE CASCADE)
ALTER TABLE tbl_rental
ADD CONSTRAINT fk_Member_id
FOREIGN KEY (Member_id)
REFERENCES tbl_member(Member_id)
ON DELETE CASCADE;

-- 6) 제약 조건 확인 (MySQL용: information_schema 사용)
SELECT
    TABLE_NAME,
    CONSTRAINT_NAME,
    CONSTRAINT_TYPE
FROM information_schema.TABLE_CONSTRAINTS
WHERE TABLE_NAME IN ('tbl_member', 'tbl_book', 'tbl_rental')
  AND CONSTRAINT_SCHEMA = DATABASE();

-- 7) FK 컬럼에 인덱스 추가 후 확인

CREATE INDEX idx_rental_member_id ON tbl_rental(Member_id);
CREATE INDEX idx_rental_book_code ON tbl_rental(Book_code);

-- 인덱스 확인
SELECT
    TABLE_NAME,
    INDEX_NAME,
    COLUMN_NAME
FROM information_schema.STATISTICS
WHERE TABLE_NAME = 'tbl_rental'
  AND TABLE_SCHEMA = DATABASE();

-- 8) 뷰 생성 및 조회

CREATE OR REPLACE VIEW ShowRental_view AS
SELECT
    r.Rental_id,
    m.Member_name,
    b.book_name
FROM tbl_rental r
INNER JOIN tbl_member m ON r.Member_id = m.Member_id
INNER JOIN tbl_book b ON r.Book_code = b.Book_code;

-- 뷰 데이터 조회
SELECT * FROM ShowRental_view;
